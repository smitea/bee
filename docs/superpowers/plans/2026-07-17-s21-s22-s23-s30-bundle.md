```rust
fn load_scheduler_config() -> bee_dsl_sql::scheduler::SchedulerConfig {
    use std::str::FromStr;
    if let Ok(s) = std::env::var("BEE_RUNTIME__SCHEDULER_POLICY") {
        return SchedulerConfig::from_str(&s)
            .unwrap_or_else(|e| {
                eprintln!("warning: invalid BEE_RUNTIME__SCHEDULER_POLICY: {e}; using default");
                SchedulerConfig::default()
            });
    }
    if let Ok(path) = std::env::var("BEE_CONFIG") {
        if let Ok(s) = std::fs::read_to_string(&path) {
            // Trivial TOML: one `[scheduler]\npolicy = "mlfq"` block.
            // Keep the parser minimal — full TOML is overkill.
            for line in s.lines() {
                let line = line.trim();
                if let Some(rest) = line.strip_prefix("policy") {
                    let rest = rest.trim().trim_start_matches('=').trim().trim_matches('"');
                    return SchedulerConfig::from_str(rest).unwrap_or_else(|e| {
                        eprintln!("warning: invalid policy in {path}: {e}; using default");
                        SchedulerConfig::default()
                    });
                }
            }
        }
    }
    SchedulerConfig::default()
}
```

```rust
    /// Safe wrapper for `secret_get`. Returns `Ok(Some(value))` if
    /// found, `Ok(None)` if not found, `Err(SdkError)` on error.
    /// The plugin receives a fresh `Vec<u8>`; the host-allocated
    /// bytes are leaked (MVP — the plugin process exits shortly).
    pub fn safe_secret_get(
        &self,
        tenant: u16,
        id: &str,
    ) -> Result<Option<Vec<u8>>, SdkError> {
        let secret_get = self
            .secret_get
            .ok_or(SdkError::HostFnMissing("secret_get"))?;
        let c_id = std::ffi::CString::new(id)
            .map_err(|e| SdkError::KvError(format!("secret_get cstring: {e}")))?;
        let mut out_value: *mut u8 = std::ptr::null_mut();
        let mut out_len: usize = 0;
        let rc = unsafe {
            secret_get(self.ctx, tenant, c_id.as_ptr(), &mut out_value, &mut out_len)
        };
        match rc {
            0 => {
                if out_value.is_null() || out_len == 0 {
                    return Ok(None);
                }
                let bytes = unsafe {
                    std::slice::from_raw_parts(out_value, out_len).to_vec()
                };
                // TODO: free via host_alloc_free (S30.x follow-up).
                Ok(Some(bytes))
            }
            1 => Ok(None),
            _ => Err(SdkError::KvError("secret_get failed".into())),
        }
    }

    /// Safe wrapper for `secret_put`. Returns `Ok(())` on success,
    /// `Err(SdkError)` on error.
    pub fn safe_secret_put(
        &self,
        tenant: u16,
        id: &str,
        value: &[u8],
    ) -> Result<(), SdkError> {
        let secret_put = self
            .secret_put
            .ok_or(SdkError::HostFnMissing("secret_put"))?;
        let c_id = std::ffi::CString::new(id)
            .map_err(|e| SdkError::KvError(format!("secret_put cstring: {e}")))?;
        let rc = unsafe {
            secret_put(self.ctx, tenant, c_id.as_ptr(), value.as_ptr(), value.len())
        };
        match rc {
            0 => Ok(()),
            _ => Err(SdkError::KvError("secret_put failed".into())),
        }
    }
```
